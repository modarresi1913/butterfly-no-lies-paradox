   931→  const [error, setError] = useState<string | null>(null);

   933→
   934→  const handleFieldChange = useCallback((field: string, value: string) => {
   935→    if (field === 'era') setEra(value);
   936→    else if (field === 'location') setLocation(value);
   937→    else if (field === 'change') setChange(value);
   938→  }, []);
   939→
   940→  const handleReset = useCallback(() => {
   941→    if (abortRef.current) abortRef.current.abort();
   942→    setStep('idle');
   943→    setEra('');
   944→    setLocation('');
   945→    setChange('');
   946→    setMagnitude('public');
   947→    setCheckpoints([]);
   948→    setFeaturedImage(null);
   949→    setError(null);

   953→    setStep('simulate');

   972→      const data = await res.json();
   973→      if (data.success) {
   974→        setCheckpoints(data.checkpoints || []);
   975→        setFeaturedImage(data.featured_image || null);
   976→        setStep('results');

   978→        throw new Error(data.error || 'Invalid server response');
   979→      }

  1001→            <Button
  1002→              variant="ghost"
  1003→              size="sm"
  1004→              onClick={handleReset}
  1005→              className="text-muted-foreground hover:text-foreground gap-1.5 text-xs"
  1006→            >
  1007→              <RotateCcw className="w-3.5 h-3.5" />
  1008→              Reset
  1009→            </Button>
