   961→  const [location, setLocation] = useState('');
   962→  const [change, setChange] = useState('');
   963→  const [magnitude, setMagnitude] = useState<Magnitude>('public');
   964→  const [checkpoints, setCheckpoints] = useState<Checkpoint[]>([]);
   965→  const [featuredImage, setFeaturedImage] = useState<string | null>(null);
   966→  const [error, setError] = useState<string | null>(null);
   967→  const abortRef = useRef<AbortController | null>(null);
   968→
   969→  const handleFieldChange = useCallback((field: string, value: string) => {
   970→    if (field === 'era') setEra(value);
   971→    else if (field === 'location') setLocation(value);
   972→    else if (field === 'change') setChange(value);
   973→  }, []);
   974→
   975→  const handleReset = useCallback(() => {
   976→    if (abortRef.current) abortRef.current.abort();
   977→    setStep('idle');
   978→    setEra('');
   979→    setLocation('');
   980→    setChange('');
   981→    setMagnitude('public');
   982→    setCheckpoints([]);
   983→    setFeaturedImage(null);
   984→    setError(null);
   985→  }, []);
   986→
   987→  const handleSimulate = useCallback(async () => {
   988→    setStep('simulate');
   989→    setError(null);
   990→
   991→    const controller = new AbortController();
   992→    abortRef.current = controller;
   993→
   994→    try {
   995→      const res = await fetch('/api/simulate', {
   996→        method: 'POST',
   997→        headers: { 'Content-Type': 'application/json' },
   998→        body: JSON.stringify({ era, location, change, magnitude }),
   999→        signal: controller.signal,
  1000→      });
  1001→
  1002→      if (!res.ok) {
  1003→        const errData = await res.json().catch(() => ({}));
  1004→        throw new Error(errData.error || 'خطایی در شبیه‌سازی رخ داد');
  1005→      }
  1006→
  1007→      const data = await res.json();
  1008→      if (data.success) {
  1009→        setCheckpoints(data.checkpoints || []);
  1010→        setFeaturedImage(data.featured_image || null);
  1011→        setStep('results');
  1012→      } else {
  1013→        throw new Error(data.error || 'پاسخ نامعتبر از سرور');
  1014→      }
  1015→    } catch (err: any) {
  1016→      if (err.name !== 'AbortError') {
  1017→        setError(err.message || 'خطای ناشناخته');
  1018→        setStep('ripple');
  1019→      }
  1020→    }
  1021→  }, [era, location, change, magnitude]);

  1031→            <ButterflyIcon className="w-7 h-7 text-amber" />
  1032→            <span className="font-bold text-sm md:text-base tracking-tight">
  1033→              The Butterfly Effect
  1034→            </span>
  1035→          </div>
  1036→          {step !== 'idle' && (
  1037→            <Button
  1038→              variant="ghost"
  1039→              size="sm"
  1040→              onClick={handleReset}
  1041→              className="text-muted-foreground hover:text-foreground gap-1.5 text-xs"
  1042→            >
  1043→              <RotateCcw className="w-3.5 h-3.5" />
  1044→              شروع مجدد
  1045→            </Button>
  1046→          )}
  1047→        </div>
  1048→      </header>
  1049→
  1050→      {/* Main Content */}
  1051→      <main className="relative z-10 flex-1 px-4 py-6 md:py-10">
  1052→        {step !== 'idle' && step !== 'simulate' && step !== 'results' && (
  1053→          <StepIndicator currentStep={step} />
  1054→        )}

  1057→          <motion.div
  1058→            initial={{ opacity: 0, y: -10 }}
  1059→            animate={{ opacity: 1, y: 0 }}
